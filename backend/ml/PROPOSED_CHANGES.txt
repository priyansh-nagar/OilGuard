def save_checkpoint(epoch, model, optimizer, best_val_loss, history):
    """Save training checkpoint after each epoch."""
    checkpoint = {
        'epoch': epoch,
        'model_state_dict': model.state_dict(),
        'optimizer_state_dict': optimizer.state_dict(),
        'best_val_loss': best_val_loss,
        'history': history,
        'random_seed': RANDOM_SEED,
        'config': {
            'batch_size': BATCH_SIZE,
            'learning_rate': LEARNING_RATE,
            'num_epochs': NUM_EPOCHS,
        }
    }
    torch.save(checkpoint, CHECKPOINT_PATH)
    print(f"  → Checkpoint saved (epoch {epoch})", flush=True)


def load_checkpoint():
    """Load training checkpoint if it exists."""
    if CHECKPOINT_PATH.exists():
        checkpoint = torch.load(CHECKPOINT_PATH)
        print(f"Found checkpoint at epoch {checkpoint['epoch']}")
        return checkpoint
    return None


def save_best_model(model, metrics):
    """Save the best model (lighter than full checkpoint)."""
    torch.save({
        'model_state_dict': model.state_dict(),
        'metrics': metrics,
    }, BEST_MODEL_PATH)
    print(f"  → Best model saved", flush=True)

REASON: Enable resume and save best model separately
